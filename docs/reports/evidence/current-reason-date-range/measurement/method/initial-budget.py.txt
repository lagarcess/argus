"""Run-local HTTP spend guard; no production code or request payload changes."""
import json, math, threading, time
from pathlib import Path
from urllib.parse import urlparse
import httpx

class SpendStop(BaseException):
    pass

class Budget:
    def __init__(self, path, prices, cap=5.0):
        self.path=Path(path); self.cap=cap; self.label='preflight'; self.lock=threading.RLock()
        self.rows=[]; self.blocked=[]; self.rates={}
        for model,data in prices.items():
            rates=[]
            for endpoint in data['endpoints']:
                p=endpoint['pricing']; rates.extend([p,*p.get('overrides',[])])
            self.rates[model]={
                'input':max(float(p.get(k,0)) for p in rates for k in ('prompt','input_cache_write','input_cache_write_1h','input_cache_read')),
                'output':max(float(p.get(k,0)) for p in rates for k in ('completion','internal_reasoning')),
                'request':max(float(p.get('request',0)) for p in rates),
            }
    def total(self):
        return sum(r['accounted_usd'] for r in self.rows)
    def save(self):
        payload={'cap_usd':self.cap,'accounted_upper_bound_usd':self.total(),
                 'reported_cost_usd':sum(r.get('reported_cost_usd',0) for r in self.rows),
                 'unknown_cost_reserved_usd':sum(r['accounted_usd'] for r in self.rows if 'reported_cost_usd' not in r),
                 'pricing_allowance_multiplier':1.1,'calls':self.rows,'blocked':self.blocked}
        self.path.write_text(json.dumps(payload,indent=2))
    def stop(self,reason):
        self.blocked.append({'label':self.label,'reason':reason});self.save();raise SpendStop(reason)
    def before(self,request):
        if request.method in ('GET','HEAD','OPTIONS'):return None
        with self.lock:
            if self.blocked:self.stop('prior_guard_failure')
            u=urlparse(str(request.url)); payload=json.loads(request.content)
            if u.netloc=='openrouter.ai' and u.path=='/api/v1/chat/completions':
                model=payload.get('model'); rate=self.rates.get(model)
                if not rate:self.stop('unpriced_model')
                if payload.get('tools') or payload.get('plugins') or payload.get('stream'):
                    self.stop('unbounded_tools_or_stream')
                limit=payload.get('max_tokens')
                if not isinstance(limit,int) or limit<=0:self.stop('missing_output_limit')
                # UTF-8 bytes conservatively upper-bound text tokens; include all
                # schema bytes and extra protocol overhead, with 10% price margin.
                reserve=((len(request.content)+1024)*rate['input']+limit*rate['output']+rate['request'])*1.1
                provider='openrouter'
            elif u.netloc=='api.perplexity.ai' and u.path=='/search':
                model=None;provider='perplexity_search';reserve=0.005*1.1*max(1,len(payload['query']) if isinstance(payload.get('query'),list) else 1)
            else:self.stop('unpriced_write_endpoint:'+u.netloc+u.path)
            if not math.isfinite(reserve) or self.total()+reserve>self.cap:
                self.stop('budget_would_exceed_cap')
            row={'attempt':len(self.rows)+1,'label':self.label,'provider':provider,'model':model,
                 'reserved_usd':reserve,'accounted_usd':reserve,'started_at':time.time()}
            self.rows.append(row);self.save();return row
    def after(self,row,response):
        if row is None:return
        with self.lock:
            row['status_code']=response.status_code
            if row['provider']=='openrouter':
                try:
                    data=response.json(); cost=data.get('usage',{}).get('cost')
                    row['served_model']=data.get('model'); row['usage']=data.get('usage')
                    if isinstance(cost,(float,int)) and math.isfinite(cost) and cost>=0:
                        if cost*1.1>row['reserved_usd']+1e-9:self.stop('provider_exceeded_reservation')
                        row['reported_cost_usd']=cost;row['accounted_usd']=cost*1.1
                except (ValueError,TypeError):pass
            row['finished_at']=time.time();self.save()
    def install(self):
        original_async=httpx.AsyncClient.send; original_sync=httpx.Client.send
        guard=self
        async def send_async(client,request,*args,**kwargs):
            row=guard.before(request)
            result=await original_async(client,request,*args,**kwargs)
            await result.aread();guard.after(row,result);return result
        def send_sync(client,request,*args,**kwargs):
            row=guard.before(request)
            result=original_sync(client,request,*args,**kwargs)
            result.read();guard.after(row,result);return result
        httpx.AsyncClient.send=send_async;httpx.Client.send=send_sync
        # Alpaca uses requests for its read-only market-data calls. Reject any
        # write that bypasses the accounted HTTP client.
        import requests
        original_request=requests.Session.request
        def request(session,method,url,*args,**kwargs):
            if method.upper() not in ('GET','HEAD','OPTIONS'):guard.stop('unaccounted_requests_write')
            return original_request(session,method,url,*args,**kwargs)
        requests.Session.request=request
