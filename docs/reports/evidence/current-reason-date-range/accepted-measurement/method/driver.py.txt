class RecordingBudget(Budget):
    def before(self,request):
        row=super().before(request)
        if row is not None and row['provider']=='openrouter':
            body=json.loads(request.content)
            row['schema_name']=body.get('response_format',{}).get('json_schema',{}).get('name')
        return row
    def after(self,row,response):
        if row is not None:
            try:data=response.json()
            except ValueError:data={'non_json_body_excerpt':response.text[:4000]}
            record={'attempt':row['attempt'],'case_id':row['label'],'provider':row['provider'],'schema_name':row.get('schema_name'),'status_code':response.status_code,'response':data}
            encoded=json.dumps(record)
            # Preserve model JSON, never credentials or headers.
            for key in keys:encoded=encoded.replace(secret_values[key],'[REDACTED_CREDENTIAL]')
            with self.lock:
                with rawpath.open('a') as stream:stream.write(encoded+'\n')
        return super().after(row,response)
